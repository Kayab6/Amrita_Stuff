## sum of EVEN numbers bw m and n

'''
#include <stdio.h>

void main(){
    int i,n,m,sum=0;
    printf("enter n values");
    scanf("%d",&n);
    printf("enter m values");
    scanf("%d",&m);

    for(int i=n;i<m;i++){
        if(i%2==0){
            sum=sum+i;
        }
        
    }
    printf("%d",sum);
}

'''
