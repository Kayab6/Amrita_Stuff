- code to print n natural numbers

- code to print even numbers and count them 
#+BEGIN_SRC c
// even numbers
# include <stdio.h>

void main(){
    int i,n,count=0;

    printf("enter n values");
    scanf("%d",&n);

    for(int i=1;i<=n;i++){
        if(i%2==0){
            printf("%d\n",i);
            count=count+1;
        }
    }

    // how would we count 
    printf("the count of even no is%d",count);
}
